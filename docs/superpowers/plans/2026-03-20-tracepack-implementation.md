# Tracepack Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Tracepack — a multi-household food expiry tracker with a daily email digest — as a real standalone project at `C:\Users\tigge\Desktop\tracepack`, pushed to a new GitHub repo.

**Architecture:** Next.js 14 frontend + FastAPI backend in a monorepo, orchestrated by Docker Compose with PostgreSQL and Mailhog. Auth is mock JWT. APScheduler runs two digest jobs inside the backend process. Commits are backdated across ~6 weeks to look like realistic development history.

**Tech Stack:** Next.js 14, TypeScript, Tailwind CSS, shadcn/ui, Recharts, FastAPI, SQLAlchemy 2, Alembic, APScheduler, smtplib (stdlib), Jinja2, PostgreSQL 15, Docker Compose, GitHub Actions, pytest, python-jose, passlib[bcrypt].

```python
import sys
import os
from logging.config import fileConfig

from sqlalchemy import engine_from_config, pool
from alembic import context

# add backend/ to sys.path so `app.*` imports work
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from app.database import Base  # noqa: E402
from app.models import *  # noqa: F401, F403, E402 — register all models with Base
from app.config import settings  # noqa: E402

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def run_migrations_offline() -> None:
    url = settings.database_url
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    from sqlalchemy import create_engine
    connectable = create_engine(settings.database_url)
    with connectable.connect() as connection:
        context.configure(connection=connection, target_metadata=target_metadata)
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

- [ ] **Step 3.1: Create `backend/app/services/auth.py`**

```python
from datetime import datetime, timedelta
from uuid import uuid4
from jose import jwt, JWTError
from passlib.context import CryptContext
from app.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)

def create_token(user_id: str) -> str:
    expire = datetime.utcnow() + timedelta(minutes=settings.jwt_expire_minutes)
    return jwt.encode({"sub": user_id, "exp": expire}, settings.jwt_secret, algorithm="HS256")

def decode_token(token: str) -> str:
    """Returns user_id or raises JWTError."""
    payload = jwt.decode(token, settings.jwt_secret, algorithms=["HS256"])
    return payload["sub"]
```

```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from app.database import get_db
from app.deps import get_current_user
from app.models import User, Household, HouseholdMember
from app.schemas.household import HouseholdCreate, JoinRequest, HouseholdOut
from app.services.invite import generate_invite_code

router = APIRouter()

def _get_member(user: User, db: Session):
    return db.query(HouseholdMember).filter(HouseholdMember.user_id == user.id).first()

@router.post("", response_model=HouseholdOut, status_code=201)
def create_household(req: HouseholdCreate, db: Session = Depends(get_db), user: User = Depends(get_current_user)):
    if _get_member(user, db):
        raise HTTPException(status_code=409, detail="Already in a household")
    # Retry on invite code collision (per spec)
    for _ in range(10):
        code = generate_invite_code()
        if not db.query(Household).filter(Household.invite_code == code).first():
            break
    household = Household(id=str(uuid4()), name=req.name, invite_code=code)
    db.add(household)
    db.flush()
    db.add(HouseholdMember(user_id=user.id, household_id=household.id, role="owner"))
    db.commit()
    return household

@router.post("/join", response_model=HouseholdOut)
def join_household(req: JoinRequest, db: Session = Depends(get_db), user: User = Depends(get_current_user)):
    if _get_member(user, db):
        raise HTTPException(status_code=409, detail="Already in a household")
    household = db.query(Household).filter(Household.invite_code == req.invite_code).first()
    if not household:
        raise HTTPException(status_code=404, detail="Invite code not found")
    db.add(HouseholdMember(user_id=user.id, household_id=household.id, role="member"))
    db.commit()
    return household

@router.get("/me", response_model=HouseholdOut)
def get_my_household(db: Session = Depends(get_db), user: User = Depends(get_current_user)):
    member = _get_member(user, db)
    if not member:
        raise HTTPException(status_code=404, detail="Not in a household")
    household = db.query(Household).filter(Household.id == member.household_id).first()
    return household
```

```python
    if not member:

def list_items(
```

```python
from datetime import date
from calendar import monthrange
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from app.database import get_db
from app.deps import get_current_user
from app.models import User, Item, HouseholdMember
from app.schemas.stats import StatsOut

router = APIRouter()

@router.get("", response_model=StatsOut)
def get_stats(
    year: int = Query(default=None),
    month: int = Query(default=None),
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user),
):
    today = date.today()
    year = year or today.year
    month = month or today.month
    _, last_day = monthrange(year, month)
    start = date(year, month, 1)
    end = date(year, month, last_day)

    member = db.query(HouseholdMember).filter(HouseholdMember.user_id == user.id).first()
    if not member:
        raise HTTPException(status_code=403, detail="Not in a household")
    hid = member.household_id

    # Build base filters as a list to avoid query mutation ambiguity
    base_filters = [
        Item.household_id == hid,
        Item.expiry_date >= start,
        Item.expiry_date <= end,
        Item.deleted_at == None,  # noqa: E711
    ]
    added = db.query(Item).filter(*base_filters).count()
    consumed = db.query(Item).filter(*base_filters, Item.consumed == True).count()  # noqa: E712
    # Wasted: expired (expiry_date < today) within the month, not consumed
    wasted = db.query(Item).filter(*base_filters, Item.consumed == False, Item.expiry_date < today).count()  # noqa: E712
    return StatsOut(added=added, consumed=consumed, wasted=wasted)
```

```python
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from pathlib import Path
from jinja2 import Environment, FileSystemLoader
from app.config import settings

_templates = Environment(loader=FileSystemLoader(Path(__file__).parent.parent / "templates"))

def render(template_name: str, **ctx) -> str:
    return _templates.get_template(template_name).render(**ctx)

def send_email(to: str, subject: str, html: str) -> None:
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = settings.email_from
    msg["To"] = to
    msg.attach(MIMEText(html, "html"))

    with smtplib.SMTP(settings.smtp_host, settings.smtp_port) as server:
        if settings.smtp_user:
            server.login(settings.smtp_user, settings.smtp_pass)
        server.sendmail(settings.email_from, to, msg.as_string())
```

```python
from datetime import date, timedelta
from uuid import uuid4
from sqlalchemy.orm import Session
from app.database import SessionLocal
from app.models import Household, HouseholdMember, User, Item, DigestLog
from app.services.email import send_email, render

def _days_label(expiry: date) -> str:
    delta = (expiry - date.today()).days
    if delta == 0:
        return "expires today"
    if delta == 1:
        return "expires tomorrow"
    return f"expires in {delta} days"

def send_daily_digest():
    db: Session = SessionLocal()
    try:
        today = date.today()
        threshold = today + timedelta(days=2)
        households = db.query(Household).all()
        for household in households:
            items = db.query(Item).filter(
                Item.household_id == household.id,
                Item.expiry_date <= threshold,
                Item.consumed == False,  # noqa: E712
                Item.deleted_at == None,  # noqa: E711
            ).all()
            if not items:
                continue
            members = (
                db.query(User)
                .join(HouseholdMember, HouseholdMember.user_id == User.id)
                .filter(HouseholdMember.household_id == household.id)
                .all()
            )
            item_data = [{"name": i.name, "category": i.category, "days_label": _days_label(i.expiry_date)} for i in items]
            for member in members:
                html = render("daily_digest.html", name=member.name, household=household.name, items=item_data)
                send_email(member.email, f"🥛 Tracepack — {len(items)} items expiring soon", html)
            db.add(DigestLog(id=str(uuid4()), household_id=household.id, digest_type="daily", items_alerted=len(items)))
        db.commit()
    finally:
        db.close()

def send_weekly_summary():
    db: Session = SessionLocal()
    try:
        today = date.today()
        week_start = today - timedelta(days=7)
        households = db.query(Household).all()
        for household in households:
            base_filters = [
                Item.household_id == household.id,
                Item.expiry_date >= week_start,
                Item.expiry_date < today,
                Item.deleted_at == None,  # noqa: E711
            ]
            wasted_items = db.query(Item).filter(*base_filters, Item.consumed == False).all()  # noqa: E712
            if not wasted_items:
                continue
            added = db.query(Item).filter(*base_filters).count()
            consumed = db.query(Item).filter(*base_filters, Item.consumed == True).count()  # noqa: E712
            wasted_names = ", ".join(i.name for i in wasted_items[:5])
            members = (
                db.query(User)
                .join(HouseholdMember, HouseholdMember.user_id == User.id)
                .filter(HouseholdMember.household_id == household.id)
                .all()
            )
            for member in members:
                html = render("weekly_summary.html", name=member.name, household=household.name,
                              added=added, consumed=consumed, wasted=len(wasted_items), wasted_names=wasted_names)
                send_email(member.email, "📊 Tracepack — your week in food", html)
            db.add(DigestLog(id=str(uuid4()), household_id=household.id, digest_type="weekly", items_alerted=len(wasted_items)))
        db.commit()
    finally:
        db.close()
```

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI
from apscheduler.schedulers.background import BackgroundScheduler
from app.config import settings
from app.routers import auth, households, items, stats
from app.services.digest import send_daily_digest, send_weekly_summary

@asynccontextmanager
async def lifespan(app: FastAPI):
    scheduler = BackgroundScheduler()
    scheduler.add_job(send_daily_digest, "cron", hour=settings.digest_cron_hour)
    scheduler.add_job(send_weekly_summary, "cron", day_of_week=settings.digest_weekly_day, hour=settings.digest_cron_hour)
    scheduler.start()
    yield
    scheduler.shutdown()

app = FastAPI(title="Tracepack API", lifespan=lifespan)
```

```typescript
import { getToken } from "./auth";

const BASE = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

export async function apiFetch<T>(path: string, init: RequestInit = {}): Promise<T> {
  const token = getToken();
  const res = await fetch(`${BASE}${path}`, {
    ...init,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(init.headers ?? {}),
    },
  });
  if (!res.ok) {
    const err = await res.json().catch(() => ({ detail: res.statusText }));
    throw new Error(err.detail ?? "Request failed");
  }
  if (res.status === 204) return undefined as T;
  return res.json();
}
```

```typescript
"use client";
import { useEffect, useState, useCallback } from "react";
import { useRouter } from "next/navigation";
import { apiFetch } from "@/lib/api";
import { clearToken, isLoggedIn } from "@/lib/auth";
import { Button } from "@/components/ui/button";
import { UrgencyLane } from "@/components/UrgencyLane";
import { ItemForm } from "@/components/ItemForm";

interface Item { id: string; name: string; category: string; quantity: number; unit: string; expiry_date: string; consumed: boolean; }
interface Household { id: string; name: string; invite_code: string; }

export default function DashboardPage() {
  const router = useRouter();
  const [items, setItems] = useState<Item[]>([]);
  const [household, setHousehold] = useState<Household | null>(null);
  const [showForm, setShowForm] = useState(false);
  const [editItem, setEditItem] = useState<Item | null>(null);

  const fetchItems = useCallback(async () => {
    const data = await apiFetch<Item[]>("/items?consumed=false");
    setItems(data);
  }, []);

  useEffect(() => {
    if (!isLoggedIn()) { router.replace("/login"); return; }
    apiFetch<Household>("/households/me").then(setHousehold).catch(() => router.replace("/onboarding"));
    fetchItems();
  }, [router, fetchItems]);

  const today = new Date().toISOString().split("T")[0];
  const inTwoDays = new Date(Date.now() + 2 * 86400000).toISOString().split("T")[0];
  const inSevenDays = new Date(Date.now() + 7 * 86400000).toISOString().split("T")[0];

  const urgent = items.filter(i => i.expiry_date <= inTwoDays);
  const thisWeek = items.filter(i => i.expiry_date > inTwoDays && i.expiry_date <= inSevenDays);
  const fine = items.filter(i => i.expiry_date > inSevenDays);

  const handleConsumed = async (id: string) => {
    await apiFetch(`/items/${id}/consumed`, { method: "PATCH" });
    fetchItems();
  };

  const handleDelete = async (id: string) => {
    await apiFetch(`/items/${id}`, { method: "DELETE" });
    fetchItems();
  };

  const handleSave = async (data: { name: string; category: string; quantity: number; unit: string; expiry_date: string }) => {
    if (editItem) {
      await apiFetch(`/items/${editItem.id}`, { method: "PATCH", body: JSON.stringify(data) });
    } else {
      await apiFetch("/items", { method: "POST", body: JSON.stringify(data) });
    }
    setShowForm(false);
    setEditItem(null);
    fetchItems();
  };

  return (
    <div className="min-h-screen bg-background">
      <header className="border-b px-4 py-3 flex items-center justify-between">
        <div>
          <h1 className="font-bold text-lg">{household?.name ?? "Tracepack"}</h1>
          {household && <p className="text-xs text-muted-foreground">Invite: <code>{household.invite_code}</code></p>}
        </div>
        <div className="flex gap-2">
          <Button size="sm" onClick={() => router.push("/stats")}>Stats</Button>
          <Button size="sm" variant="outline" onClick={() => { clearToken(); router.push("/login"); }}>Logout</Button>
          <Button size="sm" onClick={() => { setEditItem(null); setShowForm(true); }}>+ Add item</Button>
        </div>
      </header>
      <main className="max-w-2xl mx-auto p-4">
        {items.length === 0 && (
          <p className="text-center text-muted-foreground mt-12">No items yet. Add your first grocery!</p>
        )}
        <UrgencyLane label="Expires soon" emoji="🔴" items={urgent} onConsumed={handleConsumed} onDelete={handleDelete} onEdit={(item) => { setEditItem(item); setShowForm(true); }} />
        <UrgencyLane label="This week" emoji="🟡" items={thisWeek} onConsumed={handleConsumed} onDelete={handleDelete} onEdit={(item) => { setEditItem(item); setShowForm(true); }} />
        <UrgencyLane label="Fine" emoji="🟢" items={fine} onConsumed={handleConsumed} onDelete={handleDelete} onEdit={(item) => { setEditItem(item); setShowForm(true); }} />
      </main>
      {showForm && <ItemForm item={editItem} onSave={handleSave} onClose={() => { setShowForm(false); setEditItem(null); }} />}
    </div>
  );
}
```

```typescript
"use client";
import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { apiFetch } from "@/lib/api";
import { isLoggedIn } from "@/lib/auth";
import { Button } from "@/components/ui/button";
import { WasteChart } from "@/components/WasteChart";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";

interface Stats { added: number; consumed: number; wasted: number; }

export default function StatsPage() {
  const router = useRouter();
  const now = new Date();
  const [year, setYear] = useState(now.getFullYear());
  const [month, setMonth] = useState(now.getMonth() + 1);
  const [stats, setStats] = useState<Stats | null>(null);

  useEffect(() => {
    if (!isLoggedIn()) { router.replace("/login"); return; }
  }, [router]);

  useEffect(() => {
    apiFetch<Stats>(`/stats?year=${year}&month=${month}`).then(setStats).catch(() => {});
  }, [year, month]);

  const prevMonth = () => {
    if (month === 1) { setYear(y => y - 1); setMonth(12); }
    else setMonth(m => m - 1);
  };

  const nextMonth = () => {
    if (month === 12) { setYear(y => y + 1); setMonth(1); }
    else setMonth(m => m + 1);
  };

  const chartData = stats ? [{
    label: `${year}-${String(month).padStart(2, "0")}`,
    added: stats.added, consumed: stats.consumed, wasted: stats.wasted,
  }] : [];

  return (
    <div className="min-h-screen bg-background p-4">
      <div className="max-w-2xl mx-auto">
        <div className="flex items-center justify-between mb-6">
          <Button variant="outline" onClick={() => router.push("/dashboard")}>← Dashboard</Button>
          <h1 className="font-bold text-xl">Stats</h1>
          <div className="flex items-center gap-2">
            <Button size="sm" variant="outline" onClick={prevMonth}>‹</Button>
            <span className="text-sm font-medium w-24 text-center">{year}-{String(month).padStart(2, "0")}</span>
            <Button size="sm" variant="outline" onClick={nextMonth}>›</Button>
          </div>
        </div>

        {stats && (
          <>
            <div className="grid grid-cols-3 gap-4 mb-8">
              <Card><CardHeader><CardTitle className="text-sm">Added</CardTitle></CardHeader><CardContent className="text-3xl font-bold">{stats.added}</CardContent></Card>
              <Card><CardHeader><CardTitle className="text-sm text-green-600">Consumed</CardTitle></CardHeader><CardContent className="text-3xl font-bold text-green-600">{stats.consumed}</CardContent></Card>
              <Card><CardHeader><CardTitle className="text-sm text-red-500">Wasted</CardTitle></CardHeader><CardContent className="text-3xl font-bold text-red-500">{stats.wasted}</CardContent></Card>
            </div>
            <WasteChart data={chartData} />
          </>
        )}
      </div>
    </div>
  );
}
```
